# Better backup commands for dana_concrete_db
# کۆماندە باشترەکان بۆ باک ئەپ کردنی داتابەیس

# Option 1: Using SSH with proper encoding
ssh root@31.97.79.253 "mysqldump -u root -p \
  --single-transaction \
  --routines \
  --triggers \
  --events \
  --default-character-set=utf8mb4 \
  --set-charset \
  --add-drop-database \
  --add-drop-table \
  --add-drop-trigger \
  --add-drop-procedure \
  --add-drop-function \
  --add-drop-event \
  --hex-blob \
  --complete-insert \
  --extended-insert=FALSE \
  --lock-tables=FALSE \
  dana_concrete_db" > C:\xampp\htdocs\dana-concrete\database\dana_concrete_db_backup.sql

# Option 2: Simple but better command
ssh root@31.97.79.253 "mysqldump -u root -p \
  --triggers \
  --routines \
  --default-character-set=utf8mb4 \
  dana_concrete_db" > C:\xampp\htdocs\dana-concrete\database\dana_concrete_db_backup.sql

# Option 3: For Windows PowerShell
ssh root@31.97.79.253 "mysqldump -u root -p --triggers --routines --default-character-set=utf8mb4 dana_concrete_db" > C:\xampp\htdocs\dana-concrete\database\dana_concrete_db_backup.sql

# Option 4: Using scp to download existing backup
scp root@31.97.79.253:/tmp/dana_concrete_db_backup.sql C:\xampp\htdocs\dana-concrete\database\
